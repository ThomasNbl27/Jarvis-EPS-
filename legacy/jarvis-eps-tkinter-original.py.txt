def trouver_cycle(classe):
classe = classe.lower().strip()
if “pro” in classe:return "Lycée Pro"
if “gt” in classe or "général" in classe or “general” in classe:return "Lycée Général"
m = re.search(r"\d+", classe)
if not m:return ""
niveau = int(m.group())
if niveau == 6:return “Cycle 3”
elif niveau in [5, 4, 3]:return “Cycle 4”
elif niveau in [2, 1, 0]:return "Lycée Général"
return ""

def exporter_pdf():
from reportlab.platypus import (SimpleDocTemplate,Paragraph,Spacer )
from reportlab.lib.styles import (getSampleStyleSheet )
connexion = sqlite3.connect("jarvis.db")
curseur = connexion.cursor()
curseur.execute("""
SELECT date, classe, activite, contenu, competences, observations
FROM cahier_texte
""")
resultats = curseur.fetchall()
connexion.close()
document = SimpleDocTemplate("cahier_eps.pdf" )
styles = getSampleStyleSheet()
elements = []
titre = Paragraph(“Cahier de texte EPS”,styles[“Title”] )
elements.append(titre)
elements.append(Spacer(1, 20))
for ligne in resultats:
texte = f""" <b>Date :</b> {ligne[0]}<br/> <b>Classe :</b> {ligne[1]}<br/> <b>Activité :</b> {ligne[2]}<br/> <b>Contenu :</b> {ligne[3]}<br/> <b>Compétences :</b> {ligne[4]}<br/> <b>Observations :</b> {ligne[5]}<br/><br/> """
paragraphe = Paragraph(texte,styles[“BodyText”] )
elements.append(paragraphe)
elements.append(Spacer(1, 15))
document.build(elements)
messagebox.showinfo("Succès","PDF créé avec succès." )

# =========================# ALERTES ÉLÈVES# =========================
def verifier_alertes():
connexion = sqlite3.connect("jarvis.db" )
curseur = connexion.cursor()
curseur.execute("""
SELECT nom, oublis, absences, dispenses
FROM eleves
WHERE oublis >= 3 OR absences >= 3 OR dispenses >= 3
""")
alertes = curseur.fetchall()
connexion.close()
if alertes:
message = ""
for eleve in alertes:
message += (f"{eleve[0]} "f"(O:{eleve[1]} "f"A:{eleve[2]} "f"D:{eleve[3]})\n" )
messagebox.showwarning("Alertes élèves",message )
verifier_alertes()
